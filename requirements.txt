# Everything: the MCP server (requirements-mcp.txt) plus the web app and cloud providers.
-r requirements-mcp.txt

# --- Web API ---
# The web framework that turns our Python code into an API.
fastapi
# The server that actually runs FastAPI. [standard] adds nice extras like auto-reload.
uvicorn[standard]

# --- RAG: loading & splitting documents ---
# LangChain: the toolkit that glues the RAG steps together.
langchain
# The text splitters (chunking).
langchain-text-splitters
# Reads text out of PDF files.
pypdf

# --- RAG: embedding & vector storage ---
# Ollama integration (local embedding model + LLM).
langchain-ollama
# pgvector / Postgres vector store integration.
langchain-postgres
# PostgreSQL driver used by langchain-postgres (psycopg 3).
psycopg[binary]

# --- Web frontend: file uploads ---
# Lets FastAPI accept uploaded files (multipart/form-data).
python-multipart

# --- Cloud deployment: hosted model + config ---
# Google Gemini models (chat + embeddings) for the cloud version.
langchain-google-genai
# Loads a local .env file so secrets stay out of the code.
python-dotenv

# --- Local embeddings (no API rate limits, runs anywhere) ---
fastembed

# --- MCP developer tools (`mcp dev`, Inspector integration) ---
mcp[cli]>=2,<3
